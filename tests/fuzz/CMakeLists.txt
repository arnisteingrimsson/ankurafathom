add_executable(fuzz_expression expression.cpp)
target_link_libraries(fuzz_expression PRIVATE fathom_core)
add_executable(fuzz_ir_loader ir_loader.cpp)
target_link_libraries(fuzz_ir_loader PRIVATE fathom_ir)
set(FATHOM_LIBFUZZER_LIBRARY "" CACHE FILEPATH "Optional standalone libFuzzer runtime (including main)")
foreach(target fuzz_expression fuzz_ir_loader)
  if(FATHOM_LIBFUZZER_LIBRARY)
    if(NOT EXISTS "${FATHOM_LIBFUZZER_LIBRARY}")
      message(FATAL_ERROR "FATHOM_LIBFUZZER_LIBRARY does not exist")
    endif()
    target_link_libraries(${target} PRIVATE "${FATHOM_LIBFUZZER_LIBRARY}")
  else()
    target_link_options(${target} PRIVATE -fsanitize=fuzzer)
  endif()
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
endforeach()
